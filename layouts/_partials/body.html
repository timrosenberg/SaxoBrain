{{- /* Page body with Obsidian links resolved: [[File Name]] or [[File Name|Display]] points to a work or composer page. A link that matches no file stops the build, so typos never publish. Hugo has already turned straight quotes and & into entities by the time this runs, so names are compared after unescaping and with curly quotes flattened. */ -}} {{- $idx := partialCached "page-index.html" "all" -}} {{- $html := .Content | string -}} {{- $page := . -}} {{- range findRE `\[\[[^\]]+\]\]` $html -}} {{- $match := . -}} {{- $parts := split (strings.TrimSuffix "]]" (strings.TrimPrefix "[[" $match)) "|" -}} {{- $name := index $parts 0 -}} {{- $key := lower (replace (replace (replace (htmlUnescape $name) "’" "'") "‘" "'") " " " ") -}} {{- with index $idx $key -}} {{- $label := .Title | htmlEscape -}} {{- if gt (len $parts) 1 }}{{ $label = index $parts 1 }}{{ end -}} {{- $html = replace $html $match (printf `%s` .RelPermalink $label) -}} {{- else -}} {{- errorf "Link [[%s]] in %s does not match any page file. Check the spelling." $name $page.File.Path -}} {{- end -}} {{- end -}} {{- return ($html | safeHTML) -}}
