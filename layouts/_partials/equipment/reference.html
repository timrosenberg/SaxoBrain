{{- /* Keep former heading fragments on their corresponding cells when a reference becomes a table. */ -}} {{- $page := . -}} {{- $html := partial "body.html" $page | string -}} {{- $anchors := slice -}} {{- range index .Params "equipment-table-anchors" -}} {{- $label := $page.RenderString .text | plainify | htmlUnescape -}} {{- $anchors = $anchors | append (dict "id" .id "label" $label) -}} {{- end -}} {{- $used := slice -}} {{- range findRE `(?s).*?` $html -}} {{- $original := . -}} {{- $row := . -}} {{- /* Match visible text so editing emphasis does not break permanent fragments. */ -}} {{- range findRE `(?s).*?` $row -}} {{- $cell := . -}} {{- $label := $cell | plainify | htmlUnescape -}} {{- range $anchors -}} {{- if and (not (in $used .id)) (eq $label .label) -}} {{- $row = replace $row $cell (replace $cell `` (printf `` .id)) -}} {{- $used = $used | append .id -}} {{- end -}} {{- end -}} {{- end -}} {{- $html = replace $html $original $row -}} {{- end -}} {{- range $anchors -}} {{- if not (in $used .id) }}{{ errorf "Equipment table in %s has no cell for fragment %q" $page.File.Path .id }}{{ end -}} {{- end -}} {{- /* Keep former embedded fragments beside the link to their new location. */ -}} {{- range $target, $aliases := index .Params "equipment-fragment-aliases" -}} {{- $opening := findRE (printf `<[^>]+\s+id="%s"[^>]*>` $target) $html 1 -}} {{- if not $opening -}} {{- errorf "Equipment fragment aliases in %s point to missing fragment %q" $page.File.Path $target -}} {{- else -}} {{- $markers := "" -}} {{- range $aliases -}} {{- $markers = printf `%s` $markers . -}} {{- end -}} {{- $html = replace $html (index $opening 0) (printf "%s%s" (index $opening 0) $markers) -}} {{- end -}} {{- end -}} {{- return ($html | safeHTML) -}}
